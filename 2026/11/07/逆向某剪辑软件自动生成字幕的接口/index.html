函数体里只有一个 QMetaObject::activate，其余都是对参数的析构。

moc（Qt 的元对象编译器）为每个 signals: 里的函数自动生成的实现。它的作用就是通知一下”这个信号发出去了”，本身没有业务逻辑。

(连接对象临时量, 发送者, 信号函数指针, 接收者, 槽函数指针, this 调整值)

6 个节，.text、.rdata、.data、.pdata、.rsrc、.reloc

光从这里可能还是看不出是否加了壳。算算熵值：

import pefile, math, collections
pe = pefile.PE('target.exe')
for s in pe.sections:
    d = s.get_data()
    c = collections.Counter(d)
    n = len(d)
    h = -sum(v/n*math.log2(v/n) for v in c.values()) if n else 0# 计算熵值
    print(f'{s.Name.rstrip(bytes(8)).decode():10s} vsz={s.Misc_VirtualSize:>10d} ent={h:.2f}')

计算结果如下：

> python .\entropy_calc.py

壳的痕迹是“节区名异常 + 代码节熵接近 8”。x64 正常代码节熵在 5.5–6.2，数据节高于代码节是正常的，因为里面是字符串和浮点常量。
所以根据到目前为止的信息（节区名正常，入口点可反编译，无异常跳转），大概可以判断没有加壳。
（.pdata 是异常处理的，条目密集且无明显文本特征就足够表明没有什么有用信息）

def parse_cookie(s):
    """把 Cookie 字符串拆成 dict。

    bare = objstore_uri[len("objstore://"):]
    bucket, key = bare.split("/", 1)
    key = "/" + key

    base = "https:" + endpoint + "/" + bucket + key
    hdr = {"X-Objstore-Auth": auth}

    total = os.path.getsize(audio_path)
    chunks = (total + chunk_size - 1) // chunk_size

    print("[3] 初始化分片上传（共 %d 片）" % chunks)
    r = sess.post(base + "?uploads&output=json", headers=hdr, timeout=30)
    r.raise_for_status()
    j = read_json(r, "objstore init")
    if j.get("OK") != 1:
        raise RuntimeError("初始化上传失败: %s" % r.text)
    upload_id = j["upload_id"]
    print("    upload_id = %s" % upload_id)

    with open(audio_path, "rb") as f:
        for i in range(chunks):
            start = i * chunk_size
            f.seek(start)
            data = f.read(chunk_size)
            end = start + len(data)
            pn = i + 1
            url = ("%s?output=json&uploadId=%s&chunks=%d&partNumber=%d"
                   "&size=%d&start=%d&end=%d&total=%d"
                   % (base, upload_id, chunks, pn, len(data), start, end, total))
            print("    上传第 %d/%d 片  %d..%d" % (pn, chunks, start, end))
            r = sess.put(url, headers=hdr, data=data, timeout=120)
            r.raise_for_status()
            if read_json(r, "objstore part").get("OK") != 1:
                raise RuntimeError("分片 %d 上传失败: %s" % (pn, r.text))

    print("[3] 合并分片")
    url = ("%s?output=json&profile=%s&uploadId=%s&biz_id=0"
           % (base, PROFILE.replace("/", "%2F"), upload_id))
    r = sess.post(url, headers=hdr, timeout=60)
    r.raise_for_status()
    j = read_json(r, "objstore merge")
    if j.get("OK") != 1:
        raise RuntimeError("合并失败: %s" % r.text)
    print("    location = %s" % j.get("location"))
    return objstore_uri


def step_create_task(sess, objstore_uri, deviceid, csrf):
    """提交识别任务"""
    print("[4] 提交 ASR 任务")
    # 注意：params 是「字符串化」的 JSON，不是嵌套对象
    params = {
        "innet_result": False,
        "model_type": "16k_zh",
        "punc": True,
        "punc_param": {"is_itn": True, "max_punctuated_length": 25},
        "resource": objstore_uri,
        "vad": True,
        "with_word_level_result": True,
    }
    body = {
        "biz_name": BIZ_NAME,
        "model_id": MODEL_ID,
        "params": json.dumps(params, separators=(",", ":"), ensure_ascii=False),
    }
    url = "%s/x/svc/v1/task?csrf=%s&trace_id=" % (BASE, csrf)
    print("    URL = %s" % url)
    r = sess.post(url, headers=api_headers(sess, deviceid),
                  data=json.dumps(body, separators=(",", ":")).encode("utf-8"),
                  timeout=30)
    r.raise_for_status()
    j = read_json(r, "asr create task")
    print("    响应: %s" % json.dumps(j, ensure_ascii=False)[:300])
    if j.get("code") != 0:
        # 把真正发出去的东西打出来，不然猜半天
        sent = r.request.headers.get("Cookie", "")
        print("    [!] 实际发出的 Cookie 头: %s" % sent)
        m = re.search(r"csrf_token=([^;\s]+)", sent)
        print("    [!] Cookie 里的 csrf_token: %s" % (m.group(1) if m else "<没找到>"))
        raise RuntimeError("创建任务失败 code=%s message=%s"
                           % (j.get("code"), j.get("message")))
    data = j.get("data") or {}
    # mark==1 表示任务当场就出了结果
    if data.get("mark") == 1:
        return None, data.get("result")
    task_id = data.get("task_id")
    if not task_id:
        raise RuntimeError("没有拿到 task_id: %s" % json.dumps(j, ensure_ascii=False))
    print("    task_id = %s" % task_id)
    return task_id, None


def ms_to_srt(ms):
    """毫秒 -> SRT 时间戳  HH:MM:SS,mmm"""
    ms = int(ms)
    if ms < 0:
        ms = 0
    h, ms = divmod(ms, 3600000)
    m, ms = divmod(ms, 60000)
    s, ms = divmod(ms, 1000)
    return "%02d:%02d:%02d,%03d" % (h, m, s, ms)


def result_to_srt(result):
    """把接口返回的 asr_text 转成 SRT 文本

    结构来自 AsrTask::OnServiceJobDone(sub_140D8EF20)：
      {"utterances":[{"transcript":..,"start_time":毫秒,"end_time":毫秒}]}
    """
    if isinstance(result, str):
        data = json.loads(result)
    else:
        data = result or {}

    lines = []
    idx = 0
    for u in data.get("utterances", []):
        text = (u.get("transcript") or "").strip()
        if not text:
            continue
        start = int(u.get("start_time", 0))
        end = int(u.get("end_time", start))
        if end < start:
            end = start
        idx += 1
        lines.append("%d\r\n%s --> %s\r\n%s\r\n"
                     % (idx, ms_to_srt(start), ms_to_srt(end), text))
    return "\r\n".join(lines)


def step_poll(sess, task_id, deviceid, timeout_ms=300000):
    """轮询任务结果"""
    print("[5] 轮询结果")
    path = "/x/svc/v1/task/result"
    deadline = time.time() + timeout_ms / 1000.0
    while True:
        if time.time() > deadline:
            raise RuntimeError("轮询超时（客户端原版是 5 分钟）")
        url = ("%s%s?task_id=%s&model_id=%s&biz_name=%s"
               % (BASE, path, task_id, MODEL_ID, BIZ_NAME))
        r = sess.get(url, headers=api_headers(sess, deviceid), timeout=30)
        r.raise_for_status()
        j = read_json(r, "asr poll")
        if j.get("code") != 0:
            raise RuntimeError("查询失败 code=%s message=%s"
                               % (j.get("code"), j.get("message")))
        data = j.get("data") or {}
        state = data.get("state", 0)
        poll_time = data.get("poll_time", 1000)
        if state == 0:
            print("    完成")
            return data.get("result")
        if state == 1:
            print("    处理中，%s ms 后再查" % poll_time)
            time.sleep(poll_time / 1000.0)
            continue
        raise RuntimeError("任务失败 state=%s" % state)


def main():
    if len(sys.argv) < 2:
        print("用法: python asr_replay.py <视频文件> [输出txt]")
        return 1
    src = sys.argv[1]
    out_txt = sys.argv[2] if len(sys.argv) > 2 else os.path.splitext(src)[0] + ".srt"

    audio = os.path.splitext(src)[0] + ".m4a"
    cookies = parse_cookie(COOKIE)
    csrf = cookies.get("csrf_token", "")
    if not csrf:
        print("Cookie 里没有 csrf_token，检查一下 COOKIE 配置")
        return 1
    print("[0] 解析到的 cookie 键: %s" % ", ".join(sorted(cookies.keys())))
    print("[0] 将使用的 csrf(csrf_token) = %r （长度 %d）" % (csrf, len(csrf)))

    sess = requests.Session()
    sess.cookies.update(cookies)

    # 登录态自检：nav 接口会直接告诉服务端认不认这个 Cookie
    try:
        nav = sess.get("https://api.example.com/x/web-interface/nav",
                       headers={"User-Agent": "Mozilla/5.0",
                                "Referer": "https://www.example.com/"},
                       timeout=10).json()
        nd = nav.get("data") or {}
        print("[0] 登录态: isLogin=%s mid=%s" % (nd.get("isLogin"), nd.get("mid")))
    except Exception as e:
        print("[0] 登录态自检失败: %s" % e)

    deviceid = get_deviceid(sess)

    extract_audio(src, audio)
    name = md5_file(audio) + ".m4a"
    size = os.path.getsize(audio)
    print("    文件名 = %s  (%d 字节)" % (name, size))

    # 同一个音频只需传一次，把 objstore 路径缓存下来，
    # 调第 4 步的时候就不用反复传 20+MB 了
    cache_path = os.path.join(os.path.dirname(os.path.abspath(__file__)),
                              "upload_cache.json")
    cache = {}
    if os.path.exists(cache_path):
        try:
            with open(cache_path, "r", encoding="utf-8") as f:
                cache = json.load(f)
        except Exception:
            cache = {}

    if name in cache:
        objstore_uri = cache[name]
        print("[2][3] 命中缓存，跳过上传: %s" % objstore_uri)
    else:
        info = step_upload/apply(sess, name, size, deviceid)
        objstore_uri = step_upload(sess, info, audio)
        cache[name] = objstore_uri
        with open(cache_path, "w", encoding="utf-8") as f:
            json.dump(cache, f, ensure_ascii=False, indent=2)

    task_id, result = step_create_task(sess, objstore_uri, deviceid, csrf)
    if result is None:
        result = step_poll(sess, task_id, deviceid)

    print("[6] 原始结果: %s" % (result or "")[:200])
    srt = result_to_srt(result)

    with open(out_txt, "w", encoding="utf-8") as f:
        f.write(srt)
    print("[6] 字幕已写入 %s（%d 条）" % (out_txt, srt.count(" --> ")))
    return 0


if __name__ == "__main__":
    sys.exit(main())
